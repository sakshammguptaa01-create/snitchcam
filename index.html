import cv2 from ultralytics import YOLO import datetime import os # Load pretrained YOLOv8 model (trained on COCO dataset, includes 'cell phone' class) model = YOLO("yolov8n.pt") # Create folder to save snapshots of detected phones SAVE_DIR = "detected_phones" os.makedirs(SAVE_DIR, exist_ok=True) # Class ID for "cell phone" in COCO dataset PHONE_CLASS_ID = 67 def run_snitch_cam(): cap = cv2.VideoCapture(0) # 0 = default webcam if not cap.isOpened(): print("Error: Could not access webcam.") return print("Snitch Cam started. Press 'q' to quit.") while True: ret, frame = cap.read() if not ret: break results = model(frame, verbose=False)[0] phone_detected = False for box in results.boxes: cls_id = int(box.cls[0]) conf = float(box.conf[0]) if cls_id == PHONE_CLASS_ID and conf > 0.5: phone_detected = True x1, y1, x2, y2 = map(int, box.xyxy[0]) # Draw bounding box cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 0, 255), 2) label = f"Phone Detected ({conf:.2f})" cv2.putText(frame, label, (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2) if phone_detected: cv2.putText(frame, "ALERT: PHONE DETECTED!", (20, 40), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 3) # Save a snapshot with timestamp timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S") filename = os.path.join(SAVE_DIR, f"phone_{timestamp}.jpg") cv2.imwrite(filename, frame) cv2.imshow("Snitch Cam - Phone Detector", frame) if cv2.waitKey(1) & 0xFF == ord('q'): break cap.release() cv2.destroyAllWindows() if __name__ == "__main__": run_snitch_cam()
